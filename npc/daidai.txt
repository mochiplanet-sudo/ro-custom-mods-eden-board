// ==============================================================
// 單機版呆呆每日任務 (動態等級抽選 + 每日04:00重置 + 回報防誤點確認)
// ==============================================================

// 吉芬 (Geffen)
//geffen,147,127,4

// 艾爾貝塔 (Alberta)
//alberta,211,102,4

// 夢羅克
//morocc,109,96,4

//prt_fild08,180,370,4

prt_fild08,180,370,4	script	呆呆小助手	721,{

    .@now = gettimetick(2);

    // ----------------------------------------------------------
    // 1. 檢查冷卻時間 (每日 04:00 重置)
    // ----------------------------------------------------------
    if (.@now < Daidai_CD) {
        .@rem = Daidai_CD - .@now;
        mes "[呆呆小助手]";
        mes "今日的冒險者支援已經領取過囉！";
        mes "每日重置時間為 ^FF0000凌晨 04:00^000000。";
        mes "距離下次開放還剩：^777777" + (.@rem / 3600) + " 小時 " + ((.@rem % 3600) / 60) + " 分鐘^000000。";
        close;
    }

    // ----------------------------------------------------------
    // 2. 任務已完成，回報領取獎勵 (防誤點二次確認)
    // ----------------------------------------------------------
    if (Daidai_Status == 2) {
        mes "[呆呆小助手]";
        mes "太厲害了！你已成功擊殺 " + $Daidai_ReqCount + " 隻【" + Daidai_MobName$ + "】！";
        mes "挑戰任務完成，確定要現在領取獎勵了嗎？";
        mes "^FF0000(領取後將立刻開始倒數 30 分鐘加倍與限時卷軸)^000000";
        next;

        if (select("確認領取獎勵:我還沒準備好，先不要") == 2) {
            mes "[呆呆小助手]";
            mes "好的！等整理好背包、準備出發冒險時再來找我領取。";
            close;
        }

        // 檢查背包空間 (至少 1 格放租賃卷軸)
        if (checkweight(22997, 1) == 0) {
            mes "[呆呆小助手]";
            mes "你的背包空間不足，請先空出至少 1 格空間來領取租借卷軸！";
            close;
        }

        // 賦予效果 (持續 30 分鐘 = 1800 秒 / 1800000 毫秒)
        sc_start SC_EXPBOOST, 1800000, 100;   // 經驗值 +100%
        sc_start SC_JEXPBOOST, 1800000, 100;  // Job 經驗值 +100%
        sc_start SC_ITEMBOOST, 1800000, 100;  // 掉寶率 +100%
        rentitem 22997, 1800;                 // 租賃無限貪婪卷軸 (30 分鐘)

        // 計算下一個每日凌晨 04:00 的時間戳
        // 虛擬機的 20:00 = 台灣的凌晨 04:00
        .@today_4am = gettimetick(2) - (gettime(DT_HOUR) * 3600 + gettime(DT_MINUTE) * 60 + gettime(DT_SECOND)) + 72000;
        if (.@now >= .@today_4am) {
            Daidai_CD = .@today_4am + 86400; // 明天凌晨 20:00
        } else {
            Daidai_CD = .@today_4am;         // 今天凌晨 20:00
        }

        // 重置任務數據
        Daidai_Status = 0;
        Daidai_Count = 0;
        Daidai_MobID = 0;
        Daidai_MobName$ = "";

        specialeffect2 438;
        mes "[呆呆小助手]";
        mes "【祝福已生效】：";
        mes " -> ^FF0000經驗值獲得量 +100%^000000 (持續 30 分鐘)";
        mes " -> ^FF0000全魔物掉寶率 +100%^000000 (持續 30 分鐘)";
        mes " -> ^0000FF無限貪婪卷軸^000000 已送入背包 (限時 30 分鐘)";
        mes "祝你冒險順利，明日凌晨 04:00 後歡迎再來！";
        close;
    }

    // ----------------------------------------------------------
    // 3. 任務進行中 (僅顯示進度，不可放棄、不可重抽)
    // ----------------------------------------------------------
    if (Daidai_Status == 1) {
        mes "[呆呆小助手]";
        mes "你目前正在進行挑戰任務：";
        mes "目標魔物：【^0000FF" + Daidai_MobName$ + "^000000】";
        mes "目前進度：^0000FF" + Daidai_Count + " / " + $Daidai_ReqCount + "^000000 隻。";
        mes "此任務一經指派不可更換，請擊殺達標後再來找我領賞！";
        close;
    }

    // ----------------------------------------------------------
    // 4. 接取任務：依自身等級動態篩選候選池
    // ----------------------------------------------------------
    mes "[呆呆小助手]";
    mes "你好！我是每日呆呆支援助手。";
    mes "每天完成一項魔物討伐挑戰，就可以獲得 ^FF000030 分鐘雙倍經驗、雙倍掉寶與無限貪婪能力^000000！";
    next;

    if (select("我要接受今日挑戰！:暫時不需要") == 2) {
        mes "[呆呆小助手]";
        mes "隨時歡迎你來找我！祝你有美好的一天。";
        close;
    }

    // 動態計算當前等級的最高可抽怪物等級上限 (例如 15等 -> 20等以下; 53等 -> 60等以下)
    .@max_mob_lv = ((BaseLevel / 10) + 1) * 10;

    cleararray .@Pick_ID[0], 0, getarraysize(.@Pick_ID);
    cleararray .@Pick_Name$[0], "", getarraysize(.@Pick_Name$);
    .@total = 0;

    // 遍歷名冊，只要怪物等級 <= 上限門檻，就加入本次抽籤池
    for (.@i = 0; .@i < getarraysize($Daidai_Mob_ID); .@i++) {
        if ($Daidai_Mob_Lv[.@i] <= .@max_mob_lv) {
            .@Pick_ID[.@total] = $Daidai_Mob_ID[.@i];
            .@Pick_Name$[.@total] = $Daidai_Mob_Name$[.@i];
            .@total++;
        }
    }

    // 防呆兜底：如果等級極低篩不到怪，預設抓第一隻 (蒼蠅)
    if (.@total == 0) {
        .@Pick_ID[0] = $Daidai_Mob_ID[0];
        .@Pick_Name$[0] = $Daidai_Mob_Name$[0];
        .@total = 1;
    }

    // 隨機抽取 1 隻
    .@rand = rand(0, .@total - 1);
    Daidai_MobID = .@Pick_ID[.@rand];
    Daidai_MobName$ = .@Pick_Name$[.@rand];

    Daidai_Status = 1;
    Daidai_Count = 0;

    mes "[呆呆小助手]";
    mes "抽選完畢！今日指派給你的委託是：";
    mes " -> 請討伐 ^0000FF" + $Daidai_ReqCount + " 隻【" + Daidai_MobName$ + "】^000000！";
    mes "擊殺時聊天框會即時提示進度，完成後隨時回來找我領賞！";
    close;

// ==============================================================
// 全域擊殺監聽核心 (自動更新進度，支援同畫面 20 格組隊共享)
// ==============================================================
OnNPCKillEvent:
    .@pid = getcharid(1);

    if (!.@pid) {
        callsub OnCheckDaidaiProgress, killedrid;
        end;
    }

    .@killedrid = killedrid;
    getpartymember .@pid, 1;
    getpartymember .@pid, 2;
    getmapxy .@map1$, .@x1, .@y1, UNITTYPE_PC;

    for (.@i = 0; .@i < $@partymembercount; ++.@i) {
        if (isloggedin($@partymemberaid[.@i], $@partymembercid[.@i])) {
            attachrid $@partymemberaid[.@i];
            getmapxy .@map2$, .@x2, .@y2, UNITTYPE_PC;

            if (.@map1$ == .@map2$ && distance(.@x1, .@y1, .@x2, .@y2) <= 20 && Hp > 0) {
                callsub OnCheckDaidaiProgress, .@killedrid;
            }
        }
    }
    end;

OnCheckDaidaiProgress:
    .@t = getarg(0);
    if (Daidai_Status == 1 && .@t == Daidai_MobID) {
        Daidai_Count++;
        if (Daidai_Count >= $Daidai_ReqCount) {
            Daidai_Status = 2;
            dispbottom "【呆呆任務達成】" + Daidai_MobName$ + " 討伐完成 (" + $Daidai_ReqCount + "/" + $Daidai_ReqCount + ")！請回南門找呆呆領取加倍與貪婪！";
            specialeffect2 438;
        } else {
            dispbottom "【呆呆任務進度】" + Daidai_MobName$ + "：(" + Daidai_Count + "/" + $Daidai_ReqCount + ")";
        }
    }
    return;

// ==============================================================

    $Daidai_ReqCount = 30;

    // ----------------------------------------------------------
    // 【魔物資料表】：名稱、等級、ID
    // 三個陣列的項目位置必須「一一對應」！
    // ----------------------------------------------------------
    setarray $Daidai_Mob_Name$[0],
        "蒼蠅",           // Lv.5

    end;
