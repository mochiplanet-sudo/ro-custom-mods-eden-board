// ==============================================================
// 1~39委託佈告欄 (陣列架構優化版)
// ==============================================================

prt_fild08,184,377,4	script	1~39	857,{

L_MainMenu:
    .@now = gettimetick(2);

    // 計算下一個每日凌晨 04:00 的重置時間戳 (虛擬機 20:00 = 台灣 04:00)
    .@today_4am = .@now - (gettime(DT_HOUR) * 3600 + gettime(DT_MINUTE) * 60 + gettime(DT_SECOND)) + 72000;
    if (.@now >= .@today_4am) {
        .@cd_time = .@today_4am + 86400;
    } else {
        .@cd_time = .@today_4am;
    }

    // 計算當前共有幾個委託可以回報 (陣列遍歷)
    .@claim_cnt = 0;
    for (.@i = 1; .@i <= 12; ++.@i) {
        if (callsub(L_CheckClaimable, .@i)) .@claim_cnt++;
    }

    mes "[1~39委託佈告欄]";
    mes "這裡張貼著適合等級 1～39 冒險者接取的委託懸賞。";
    mes "完成委託後，重置時間為 ^FF0000每日凌晨 04:00^000000。";
    next;

    .@claim_btn$ = (.@claim_cnt > 0 ? "^007700【★ 一鍵回報所有已完成委託 (" + .@claim_cnt + "個) ★】^000000" : "^777777【一鍵回報 (目前無可回報任務)】^000000");

    switch(select(.@claim_btn$ + ":【斐揚原野 委託】:【普隆德拉區域 委託】:【斐揚迷宮 委託】:【伊斯魯得島 委託】:【克魔島密穴 委託】:離開")) {

        // ======================================================
        // 【一鍵批次回報】
        // ======================================================
        case 1:
            if (.@claim_cnt == 0) {
                mes "[1~39佈告欄]";
                mes "目前沒有任何已完成的委託可以領賞喔！";
                next;
                goto L_MainMenu;
            }

            .@total_bexp = 0;
            .@total_jexp = 0;

            for (.@i = 1; .@i <= 12; ++.@i) {
                if (callsub(L_CheckClaimable, .@i)) {
                    // 若有需要繳交的道具則扣除
                    if ($EB_ItemID[.@i] > 0) {
                        delitem $EB_ItemID[.@i], $EB_ItemCnt[.@i];
                    }
                    .@total_bexp += $EB_BExp[.@i];
                    .@total_jexp += $EB_JExp[.@i];

                    // 狀態與計數歸零，寫入 CD
                    EB_Status[.@i] = 0;
                    EB_M1[.@i] = 0;
                    EB_M2[.@i] = 0;
                    EB_CD[.@i] = .@cd_time;
                }
            }

            getexp .@total_bexp, .@total_jexp;
            specialeffect2 438;
            mes "[1~39佈告欄]";
            mes "【一鍵回報完成！】";
            mes "本次共結算 ^0000FF" + .@claim_cnt + "^000000 個委託！";
            mes "獲得 Base 經驗值：^FF0000+" + .@total_bexp + "^000000";
            mes "獲得 Job 經驗值：^FF0000+" + .@total_jexp + "^000000";
            mes "回報的委託將於明日凌晨 04:00 重置。";
            next;
            goto L_MainMenu;

        case 2: callsub L_RegionMenu, 1; goto L_MainMenu; // 斐揚原野
        case 3: callsub L_RegionMenu, 2; goto L_MainMenu; // 普隆德拉
        case 4: callsub L_RegionMenu, 3; goto L_MainMenu; // 斐揚迷宮
        case 5: callsub L_RegionMenu, 4; goto L_MainMenu; // 伊斯魯得島
        case 6: callsub L_RegionMenu, 5; goto L_MainMenu; // 克魔島密穴
        case 7: close;
    }
    end;

// --------------------------------------------------------------
// 分區選單生成子程序 (依分區自動動態組裝選單)
// --------------------------------------------------------------
L_RegionMenu:
    .@reg = getarg(0);

    while(1) {
        .@now = gettimetick(2);
        .@menu$ = "";
        cleararray .@reg_tasks[0], 0, getarraysize(.@reg_tasks);
        .@cnt = 0;

        for (.@i = 1; .@i <= 12; ++.@i) {
            if ($EB_Region[.@i] == .@reg) {
                .@reg_tasks[.@cnt] = .@i;
                .@cnt++;

                // 判斷按鈕狀態前綴
                if (.@now < EB_CD[.@i]) {
                    .@prefix$ = "^777777【CD " + ((EB_CD[.@i] - .@now) / 3600) + "h】^000000";
                } else if (callsub(L_CheckClaimable, .@i)) {
                    .@prefix$ = "^007700【可領賞】^000000";
                } else if (EB_Status[.@i] == 1) {
                    if ($EB_M2_Req[.@i] == 0 && $EB_ItemID[.@i] == 0) {
                        .@prefix$ = "^0000FF【進行中 " + EB_M1[.@i] + "/" + $EB_M1_Req[.@i] + "】^000000";
                    } else {
                        .@prefix$ = "^0000FF【進行中】^000000";
                    }
                } else {
                    .@prefix$ = "【接取】";
                }
                .@menu$ = .@menu$ + .@prefix$ + $EB_MenuName$[.@i] + ":";
            }
        }
        .@menu$ = .@menu$ + "返回上一層";

        .@sel = select(.@menu$);
        if (.@sel == .@cnt + 1) return;

        .@task_idx = .@reg_tasks[.@sel - 1];
        callsub L_TaskDetail, .@task_idx, .@cd_time;
    }
    return;

// --------------------------------------------------------------
// 單一任務對話與交互核心
// --------------------------------------------------------------
L_TaskDetail:
    .@idx = getarg(0);
    .@cd_time = getarg(1);
    .@now = gettimetick(2);

    mes "[" + $EB_Title$[.@idx] + "] (需求 Lv " + $EB_ReqLv[.@idx] + ")";
    mes $EB_Desc$[.@idx];
    next;

    // 1. 冷卻中判定
    if (.@now < EB_CD[.@idx]) {
        .@rem = EB_CD[.@idx] - .@now;
        mes "此委託冷卻中，剩餘：" + (.@rem / 3600) + " 小時 " + ((.@rem % 3600) / 60) + " 分。";
    }
    // 2. 領賞判定
    else if (callsub(L_CheckClaimable, .@idx)) {
        if ($EB_ItemID[.@idx] > 0) {
            delitem $EB_ItemID[.@idx], $EB_ItemCnt[.@idx];
        }
        getexp $EB_BExp[.@idx], $EB_JExp[.@idx];
        EB_Status[.@idx] = 0;
        EB_M1[.@idx] = 0;
        EB_M2[.@idx] = 0;
        EB_CD[.@idx] = .@cd_time;
        specialeffect2 249;

        if ($EB_ItemID[.@idx] > 0) {
            mes "委託達成！收繳 " + $EB_ItemCnt[.@idx] + " 個" + getitemname($EB_ItemID[.@idx]) + "，進入冷卻。";
        } else {
            mes "委託達成！已發放經驗並進入冷卻。";
        }
    }
    // 3. 進行中進度顯示
    else if (EB_Status[.@idx] == 1) {
        mes callsub(L_BuildProgressStr, .@idx);
    }
    // 4. 接取判定
    else {
        if (BaseLevel < $EB_ReqLv[.@idx]) {
            mes "等級不足 " + $EB_ReqLv[.@idx] + " 級，無法接取！";
        } else if (select("接取此任務:暫時不接") == 1) {
            EB_Status[.@idx] = 1;
            EB_M1[.@idx] = 0;
            EB_M2[.@idx] = 0;
            if (.@idx == 8) mes "已接取委託！請前往斐揚迷宮洞穴。";
            else if (.@idx == 10) mes "已接取委託！請前往克魔島密穴。";
            else mes "已接取委託！";
        }
    }
    next;
    return;

// --------------------------------------------------------------
// 工具子程序：判斷是否可回報
// --------------------------------------------------------------
L_CheckClaimable:
    .@i = getarg(0);
    if ($EB_ItemID[.@i] > 0) {
        if (EB_Status[.@i] == 1 && EB_M1[.@i] >= $EB_M1_Req[.@i] && countitem($EB_ItemID[.@i]) >= $EB_ItemCnt[.@i]) return 1;
    } else {
        if (EB_Status[.@i] == 2) return 1;
    }
    return 0;

// --------------------------------------------------------------
// 工具子程序：組合進行中進度文字
// --------------------------------------------------------------
L_BuildProgressStr:
    .@i = getarg(0);
    .@str$ = "[進度] " + $EB_M1_Name$[.@i] + "：(" + EB_M1[.@i] + "/" + $EB_M1_Req[.@i] + ")";
    if ($EB_M2_Req[.@i] > 0) {
        .@str$ = .@str$ + "，" + $EB_M2_Name$[.@i] + "：(" + EB_M2[.@i] + "/" + $EB_M2_Req[.@i] + ")";
    }
    if ($EB_ItemID[.@i] > 0) {
        .@str$ = .@str$ + "，" + getitemname($EB_ItemID[.@i]) + "：(" + countitem($EB_ItemID[.@i]) + "/" + $EB_ItemCnt[.@i] + ")";
    }
    return .@str$ + "。";

// ==============================================================
// 全域擊殺監聽核心 (自動更新任務計數與跳字提醒，支援 20 格隊友共享)
// ==============================================================
OnNPCKillEvent:
    .@pid = getcharid(1);

    if (!.@pid) {
        callsub OnCheckTaskProgress, killedrid;
        end;
    }

    .@killedrid = killedrid;
    getpartymember .@pid, 1;
    getpartymember .@pid, 2;
    getmapxy .@map1$, .@x1, .@y1, UNITTYPE_PC;

    for (.@i = 0; .@i < $@partymembercount; ++.@i) {
        if (isloggedin($@partymemberaid[.@i], $@partymembercid[.@i])) {
            attachrid $@partymemberaid[.@i];
            getmapxy .@map2$, .@x2, .@y2, UNITTYPE_PC;

            if (.@map1$ == .@map2$ && distance(.@x1, .@y1, .@x2, .@y2) <= 20 && Hp > 0) {
                callsub OnCheckTaskProgress, .@killedrid;
            }
        }
    }
    end;

// --------------------------------------------------------------
// 任務擊殺判定子程序 (陣列查表匹配)
// --------------------------------------------------------------
OnCheckTaskProgress:
    .@target = getarg(0);

    for (.@i = 1; .@i <= 12; ++.@i) {
        if (EB_Status[.@i] != 1) continue;

        .@updated = 0;
        // 匹配魔物 1
        if (.@target == $EB_M1_ID[.@i] && EB_M1[.@i] < $EB_M1_Req[.@i]) {
            EB_M1[.@i]++;
            .@updated = 1;
        }
        // 匹配魔物 2
        else if ($EB_M2_Req[.@i] > 0 && .@target == $EB_M2_ID[.@i] && EB_M2[.@i] < $EB_M2_Req[.@i]) {
            EB_M2[.@i]++;
            .@updated = 1;
        }

        if (.@updated) {
            // 檢查是否擊殺全滿 (無道具任務直接標記為 Status 2 可領賞)
            if ($EB_ItemID[.@i] == 0) {
                if (EB_M1[.@i] >= $EB_M1_Req[.@i] && ($EB_M2_Req[.@i] == 0 || EB_M2[.@i] >= $EB_M2_Req[.@i])) {
                    EB_Status[.@i] = 2;
                    dispbottom "【任務達成】" + $EB_Title$[.@i] + " 完成！請回看板領賞。";
                    specialeffect2 438;
                    continue;
                }
            }

            // 聊天框跳字反饋
            .@msg$ = "【" + $EB_Title$[.@i] + "】" + $EB_M1_Name$[.@i] + "：(" + EB_M1[.@i] + "/" + $EB_M1_Req[.@i] + ")";
            if ($EB_M2_Req[.@i] > 0) {
                .@msg$ = .@msg$ + "，" + $EB_M2_Name$[.@i] + "：(" + EB_M2[.@i] + "/" + $EB_M2_Req[.@i] + ")";
            }
            if ($EB_ItemID[.@i] > 0) {
                .@msg$ = .@msg$ + "，" + getitemname($EB_ItemID[.@i]) + "：(" + countitem($EB_ItemID[.@i]) + "/" + $EB_ItemCnt[.@i] + ")";
            }
            dispbottom .@msg$;
        }
    }
    return;

// ==============================================================
